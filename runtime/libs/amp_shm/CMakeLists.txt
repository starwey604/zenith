# A sibling checkout is the default; callers may provide RPSHM_SOURCE_DIR.
set(RPSHM_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../rpshm" CACHE PATH "RPShm source checkout")
if(NOT EXISTS "${RPSHM_SOURCE_DIR}/include/rpshm/rpshm.h")
    message(FATAL_ERROR "RPShm checkout missing: set RPSHM_SOURCE_DIR")
endif()
if(NOT TARGET rpshm)
    add_subdirectory("${RPSHM_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/rpshm")
endif()
add_library(amp_shm INTERFACE)
target_link_libraries(amp_shm INTERFACE rpshm::rpshm)
target_include_directories(amp_shm INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(BUILD_TESTING)
    add_executable(amp-shm-transport-test tests/transport_test.cpp)
    target_link_libraries(amp-shm-transport-test PRIVATE amp_shm Threads::Threads)
    target_compile_options(amp-shm-transport-test PRIVATE -Wall -Wextra -Werror)
    if(NOT CMAKE_CROSSCOMPILING)
        add_test(NAME amp-shm-transport COMMAND amp-shm-transport-test)
        set_tests_properties(amp-shm-transport PROPERTIES TIMEOUT 30)
    endif()
endif()
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME amp-shm-trace-report COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/trace_report_test.py)
endif()
if(BUILD_TESTING)
    add_executable(amp-state-receiver-test tests/receiver_test.cpp)
    target_link_libraries(amp-state-receiver-test PRIVATE amp_shm)
    target_compile_options(amp-state-receiver-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    if(NOT CMAKE_CROSSCOMPILING)
        add_test(NAME amp-state-receiver-adapter COMMAND amp-state-receiver-test)
    endif()
endif()
